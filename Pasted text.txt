// We only care about successful captured payments for now.
if event == "payment.captured" {

    let payment = webhook
        .get("payload")
        .and_then(|v| v.get("payment"))
        .and_then(|v| v.get("entity"));

    if let Some(payment) = payment {

        let payment_id = payment
            .get("id")
            .and_then(|v| v.as_str())
            .unwrap_or("");

        let order_id = payment
            .get("order_id")
            .and_then(|v| v.as_str())
            .unwrap_or("");

        let amount = payment
            .get("amount")
            .and_then(|v| v.as_i64())
            .unwrap_or(0);

        let currency = payment
            .get("currency")
            .and_then(|v| v.as_str())
            .unwrap_or("");

        let email = payment
            .get("email")
            .and_then(|v| v.as_str())
            .unwrap_or("")
            .trim()
            .to_lowercase();

        let contact = payment
            .get("contact")
            .and_then(|v| v.as_str())
            .unwrap_or("");

        println!("========================================");
        println!("RAZORPAY PAYMENT CAPTURED");
        println!("Event ID: {}", event_id);
        println!("Payment ID: {}", payment_id);
        println!("Order ID: {}", order_id);
        println!("Amount: {}", amount);
        println!("Currency: {}", currency);
        println!("Email: {}", email);
        println!("Contact: {}", contact);
        println!("========================================");

        // ------------------------------------------------------------
        // Check that this is the correct DNC Flow Premium payment
        // ₹120 = 12000 paise
        // ------------------------------------------------------------

        if amount != 12000 || currency != "INR" {
            eprintln!(
                "Razorpay payment rejected: unexpected amount/currency: {} {}",
                amount,
                currency
            );

            send_response(
                &mut client,
                "200 OK",
                r#"{"success":true}"#,
                &request_id,
                start,
            )
            .await;

            return;
        }

        // ------------------------------------------------------------
        // Find the DNC Flow user by the email entered at checkout
        // ------------------------------------------------------------

        let username = match db::get_username_by_email(
            &db_client,
            &email,
            "money_assistant",
        )
        .await
        {
            Ok(Some(username)) => username,

            Ok(None) => {
                eprintln!(
                    "Razorpay payment received but no DNC Flow user found for email: {}",
                    email
                );

                send_response(
                    &mut client,
                    "200 OK",
                    r#"{"success":true}"#,
                    &request_id,
                    start,
                )
                .await;

                return;
            }

            Err(e) => {
                eprintln!(
                    "Failed to find DNC Flow user by email: {}",
                    e
                );

                send_response(
                    &mut client,
                    "500 Internal Server Error",
                    r#"{"error":"Database error"}"#,
                    &request_id,
                    start,
                )
                .await;

                return;
            }
        };

        println!(
            "Razorpay payment matched DNC Flow user: {}",
            username
        );

        // ------------------------------------------------------------
        // Prevent duplicate webhook/payment processing
        // ------------------------------------------------------------

        match db::payment_exists(&db_client, payment_id).await {
            Ok(true) => {
                println!(
                    "Razorpay payment already processed: {}",
                    payment_id
                );

                send_response(
                    &mut client,
                    "200 OK",
                    r#"{"success":true}"#,
                    &request_id,
                    start,
                )
                .await;

                return;
            }

            Ok(false) => {}

            Err(e) => {
                eprintln!(
                    "Failed to check existing Razorpay payment: {}",
                    e
                );

                send_response(
                    &mut client,
                    "500 Internal Server Error",
                    r#"{"error":"Database error"}"#,
                    &request_id,
                    start,
                )
                .await;

                return;
            }
        }

        // ------------------------------------------------------------
        // Activate DNC Flow Premium
        // ------------------------------------------------------------

        match db::activate_premium(
            &db_client,
            &username,
            "money_assistant",
            order_id,
            payment_id,
            amount as i32,
            currency,
        )
        .await
        {
            Ok(true) => {
                println!(
                    "DNC Flow Premium activated for user: {}",
                    username
                );
            }

            Ok(false) => {
                eprintln!(
                    "Failed to activate Premium: user not found: {}",
                    username
                );
            }

            Err(e) => {
                eprintln!(
                    "Failed to activate Premium for {}: {}",
                    username,
                    e
                );

                send_response(
                    &mut client,
                    "500 Internal Server Error",
                    r#"{"error":"Database error"}"#,
                    &request_id,
                    start,
                )
                .await;

                return;
            }
        }
    }
}
